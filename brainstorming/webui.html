// Node.js Backend with Git Integration
const express = require('express');
const { spawn } = require('node-pty');
const WebSocket = require('ws');
const cors = require('cors');
const { v4: uuidv4 } = require('uuid');
const path = require('path');
const fs = require('fs-extra');
const { exec } = require('child_process');
const { promisify } = require('util');

const execAsync = promisify(exec);

class RepositoryProcessManager {
    constructor() {
        this.processes = new Map();
        this.clients = new Map();
        this.workspacesDir = process.env.WORKSPACES_DIR || '/tmp/claude-workspaces';
        
        // Ensure workspaces directory exists
        fs.ensureDirSync(this.workspacesDir);
    }

    async createRepositorySession(config) {
        const sessionId = uuidv4();
        const workDir = path.join(this.workspacesDir, sessionId);
        
        const sessionInfo = {
            id: sessionId,
            name: config.name || `Session ${sessionId.slice(0, 8)}`,
            repoUrl: config.repoUrl,
            branch: config.branch || 'main',
            workDir,
            output: [],
            maxLines: 2000,
            status: 'initializing',
            createdAt: new Date(),
            pid: null,
            claudeArgs: config.claudeArgs || []
        };

        try {
            // Store session info early
            this.processes.set(sessionId, sessionInfo);
            
            // Broadcast initial status
            this.broadcast(sessionId, {
                type: 'status',
                processId: sessionId,
                status: 'cloning',
                message: 'Cloning repository...'
            });

            // Create workspace directory
            await fs.ensureDir(workDir);
            
            // Clone the repository
            await this.cloneRepository(sessionInfo);
            
            // Checkout the specified branch
            await this.checkoutBranch(sessionInfo);
            
            // Start Claude Code in the repository directory
            await this.startClaude(sessionInfo);
            
            return {
                success: true,
                session: {
                    id: sessionInfo.id,
                    name: sessionInfo.name,
                    repoUrl: sessionInfo.repoUrl,
                    branch: sessionInfo.branch,
                    pid: sessionInfo.pid,
                    status: sessionInfo.status,
                    workDir: sessionInfo.workDir
                }
            };
        } catch (error) {
            sessionInfo.status = 'error';
            sessionInfo.error = error.message;
            
            this.broadcast(sessionId, {
                type: 'error',
                processId: sessionId,
                message: error.message
            });
            
            return {
                success: false,
                error: error.message
            };
        }
    }

    async cloneRepository(sessionInfo) {
        const { repoUrl, workDir, id } = sessionInfo;
        
        this.addOutput(id, `Cloning repository: ${repoUrl}\n`, 'git');
        this.addOutput(id, `Destination: ${workDir}\n`, 'system');
        
        try {
            // Clone with progress
            const { stdout, stderr } = await execAsync(
                `git clone --progress "${repoUrl}" "${workDir}" 2>&1`
            );
            
            this.addOutput(id, stdout || stderr, 'git');
            this.addOutput(id, 'Repository cloned successfully\n', 'success');
            
        } catch (error) {
            this.addOutput(id, `Clone failed: ${error.message}\n`, 'error');
            throw new Error(`Failed to clone repository: ${error.message}`);
        }
    }

    async checkoutBranch(sessionInfo) {
        const { branch, workDir, id } = sessionInfo;
        
        this.addOutput(id, `Checking out branch: ${branch}\n`, 'git');
        
        try {
            // First fetch all branches
            await execAsync(`cd "${workDir}" && git fetch --all`);
            
            // Try to checkout the branch
            const { stdout, stderr } = await execAsync(
                `cd "${workDir}" && git checkout "${branch}" 2>&1`
            );
            
            this.addOutput(id, stdout || stderr, 'git');
            
            // Get current branch info
            const { stdout: branchInfo } = await execAsync(
                `cd "${workDir}" && git rev-parse --abbrev-ref HEAD`
            );
            
            this.addOutput(id, `Current branch: ${branchInfo}`, 'success');
            
        } catch (error) {
            // If branch doesn't exist, create it
            if (error.message.includes('did not match any file')) {
                try {
                    await execAsync(`cd "${workDir}" && git checkout -b "${branch}"`);
                    this.addOutput(id, `Created new branch: ${branch}\n`, 'success');
                } catch (createError) {
                    throw new Error(`Failed to create branch: ${createError.message}`);
                }
            } else {
                this.addOutput(id, `Checkout failed: ${error.message}\n`, 'error');
                throw new Error(`Failed to checkout branch: ${error.message}`);
            }
        }
    }

    async startClaude(sessionInfo) {
        const { workDir, id, claudeArgs } = sessionInfo;
        
        this.addOutput(id, '\n═══════════════════════════════════════\n', 'system');
        this.addOutput(id, 'Starting Claude Code...\n', 'system');
        this.addOutput(id, `Working directory: ${workDir}\n`, 'system');
        this.addOutput(id, '═══════════════════════════════════════\n\n', 'system');
        
        try {
            // Prepare Claude arguments
            const args = ['--dangerously-skip-permissions', ...claudeArgs];
            
            // Start Claude Code with PTY
            const ptyProcess = spawn('claude', args, {
                name: 'xterm-color',
                cols: 100,
                rows: 40,
                cwd: workDir,
                env: {
                    ...process.env,
                    TERM: 'xterm-256color'
                }
            });

            sessionInfo.process = ptyProcess;
            sessionInfo.pid = ptyProcess.pid;
            sessionInfo.status = 'running';

            // Handle output
            ptyProcess.on('data', (data) => {
                this.addOutput(id, data.toString(), 'stdout');
            });

            // Handle exit
            ptyProcess.on('exit', (code) => {
                sessionInfo.status = 'exited';
                sessionInfo.exitCode = code;
                sessionInfo.endedAt = new Date();
                
                this.addOutput(id, `\nClaude Code exited with code ${code}\n`, 'system');
                
                this.broadcast(id, {
                    type: 'process_exit',
                    processId: id,
                    code
                });
            });

            this.addOutput(id, `Claude Code started (PID: ${ptyProcess.pid})\n`, 'success');
            
        } catch (error) {
            this.addOutput(id, `Failed to start Claude: ${error.message}\n`, 'error');
            throw new Error(`Failed to start Claude Code: ${error.message}`);
        }
    }

    addOutput(id, data, type = 'stdout') {
        const sessionInfo = this.processes.get(id);
        if (!sessionInfo) return;
        
        const output = {
            timestamp: new Date(),
            data,
            type
        };
        
        sessionInfo.output.push(output);
        
        // Trim buffer if too large
        if (sessionInfo.output.length > sessionInfo.maxLines) {
            sessionInfo.output.shift();
        }
        
        // Broadcast to clients
        this.broadcast(id, {
            type: 'output',
            processId: id,
            data,
            outputType: type
        });
    }

    async cleanupSession(id) {
        const sessionInfo = this.processes.get(id);
        if (!sessionInfo) return;
        
        // Stop the process if running
        if (sessionInfo.process && sessionInfo.status === 'running') {
            sessionInfo.process.kill();
        }
        
        // Remove the workspace directory
        try {
            await fs.remove(sessionInfo.workDir);
            this.addOutput(id, `Cleaned up workspace: ${sessionInfo.workDir}\n`, 'system');
        } catch (error) {
            console.error(`Failed to cleanup workspace: ${error.message}`);
        }
        
        // Remove from processes map
        this.processes.delete(id);
    }

    // ... rest of the methods (sendInput, broadcast, etc.) ...
}

// Express API Setup
const app = express();
const manager = new RepositoryProcessManager();

app.use(cors());
app.use(express.json());

// API Endpoints
app.post('/api/sessions', async (req, res) => {
    const result = await manager.createRepositorySession(req.body);
    res.json(result);
});

app.get('/api/sessions', (req, res) => {
    const sessions = [];
    manager.processes.forEach(session => {
        sessions.push({
            id: session.id,
            name: session.name,
            repoUrl: session.repoUrl,
            branch: session.branch,
            status: session.status,
            pid: session.pid,
            workDir: session.workDir,
            createdAt: session.createdAt
        });
    });
    res.json(sessions);
});

app.delete('/api/sessions/:id', async (req, res) => {
    await manager.cleanupSession(req.params.id);
    res.json({ success: true });
});

// ... rest of the API endpoints ...

app.listen(3000, () => {
    console.log('Repository Process Manager running on port 3000');
});
